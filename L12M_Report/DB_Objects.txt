CREATE   PROCEDURE [_report].[SP_VIOLATIONS_L12M]
    @P_Export_Start_Date DATE = NULL,
    @P_Generate_Full_Refresh BIT = 0,
    @P_Through_Date DATE=NULL
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
    SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
    /* MAP_L12M_STAGE_BEFORE_TARGET_LOCKS_V1 */
    IF @@TRANCOUNT<>0 THROW 51000,N'Use no ambient transaction. The old initial L12M publisher is unsupported.',1;
    DECLARE @BusinessToday DATE=CONVERT(DATE,SYSUTCDATETIME() AT TIME ZONE N'UTC' AT TIME ZONE N'Eastern Standard Time');
    DECLARE @ThroughDate DATE=COALESCE(@P_Through_Date,DATEADD(DAY,-1,@BusinessToday));
    DECLARE @StartDate DATE=(SELECT Week_Begin_Date_Alt FROM HP_MAP_Prod._final.Dim_Date WHERE Full_Date=DATETRUNC(MONTH,DATEADD(MONTH,-12,@BusinessToday)));
    DECLARE @ExportStartDate DATE=CASE WHEN @P_Export_Start_Date IS NOT NULL THEN @P_Export_Start_Date
        WHEN @P_Generate_Full_Refresh=1 THEN @StartDate
        ELSE (SELECT Week_Begin_Date_Alt FROM HP_MAP_Prod._final.Dim_Date WHERE Full_Date=DATETRUNC(WEEK,DATEADD(WEEK,-2,@BusinessToday))) END;
    IF @StartDate IS NULL OR @ExportStartDate IS NULL OR @ExportStartDate<@StartDate OR @ExportStartDate>@ThroughDate OR @ThroughDate>=@BusinessToday
        THROW 51000,N'Invalid L12M calendar/export/through-date window.',1;
    IF (SELECT COUNT_BIG(*) FROM HP_MAP_Prod.sys.columns WHERE object_id=OBJECT_ID(N'HP_MAP_Prod._report.Violations_L12M'))<>85
     OR EXISTS(SELECT 1 FROM
         (SELECT name,ROW_NUMBER() OVER(ORDER BY column_id) AS Column_Ordinal FROM HP_MAP_Prod.sys.columns
          WHERE object_id=OBJECT_ID(N'HP_MAP_Prod._report.Violations_L12M')) AS live_col
         WHERE NOT EXISTS(SELECT 1 FROM (VALUES (1,N'Country'),(2,N'Validation Status'),(3,N'Validation Status Weekly'),(4,N'Fiscal Week Number'),(5,N'Week'),(6,N'Date'),(7,N'Product Category'),(8,N'Product Subcategory'),(9,N'Product Subcategory Enforcement'),(10,N'Fact ID'),(11,N'SKU'),(12,N'ASIN'),(13,N'Product_Description'),(14,N'Account'),(15,N'Seller Name'),(16,N'Storefront Name(s)'),(17,N'Location ID'),(18,N'Seller Status'),(19,N'Termination Date'),(20,N'Seller Penalty Type'),(21,N'Reporting Group'),(22,N'Market'),(23,N'Log-in Type'),(24,N'Price Location'),(25,N'MAP'),(26,N'Advertised Price'),(27,N'Eligible IR'),(28,N'Lowest Possible Price'),(29,N'Validation_Status_Raw'),(30,N'Letter Type'),(31,N'MAP Status Detail'),(32,N'Letter Delivered Date'),(33,N'Notification Generation Date'),(34,N'Buffer Period Start'),(35,N'Buffer Period End'),(36,N'Product Suspension Start'),(37,N'Product Suspension End'),(38,N'Penalty Period Start'),(39,N'Penalty Period End'),(40,N'Screenshot_URL'),(41,N'DGF Penalty'),(42,N'Notes'),(43,N'Tier_Pass_Through'),(44,N'Category Threshold Highlight'),(45,N'Category Threshold Alert'),(46,N'HP_Rep1'),(47,N'HP_Rep2'),(48,N'HP_Rep3'),(49,N'HP_Rep4'),(50,N'HP_Rep5'),(51,N'HP_Rep6'),(52,N'Email1'),(53,N'Email2'),(54,N'Email3'),(55,N'Email4'),(56,N'Email5'),(57,N'Email6'),(58,N'Selector'),(59,N'Matching Flag'),(60,N'G20 Sel/Cat Flag'),(61,N'Weekly Violation Selector'),(62,N'Validation Status New'),(63,N'RunningTotal'),(64,N'UPC'),(65,N'Bundle Flag'),(66,N'Minimum Violation Date'),(67,N'Minimum Listing Date'),(68,N'Most Recent Listing Date'),(69,N'Date Requested'),(70,N'New SKU Flag'),(71,N'Top_Reseller_Flag'),(72,N'Latest Courtesy Alert Date'),(73,N'Login_Type_ID'),(74,N'Strikethrough Flag'),(75,N'Weekly_Seller_Rank'),(76,N'Price Increase SKU Flag'),(77,N'Content Provider'),(78,N'Partner_ID'),(79,N'Violation Type'),(80,N'Product Category Latest Courtesy Alert Date'),(81,N'Monitoring Period Start'),(82,N'Monitoring Period End'),(83,N'Consequence Penalty Period Start'),(84,N'Consequence Penalty Period End'),(85,N'Flag_Warning_Or_Monitoring')) AS expected_col(Column_Ordinal,Column_Name)
                          WHERE expected_col.Column_Ordinal=live_col.Column_Ordinal AND expected_col.Column_Name COLLATE Latin1_General_100_BIN2=live_col.name COLLATE Latin1_General_100_BIN2))
        THROW 51000,N'L12M target column names/order differ from the reviewed contract; no publication.',1;
    DECLARE @Lock INT,@USLock BIT=0,@CALock BIT=0,@ReportLock BIT=0;
    BEGIN TRY
    EXEC @Lock=HP_MAP_Prod.sys.sp_getapplock @Resource=N'MAP_Operational_Writer',@LockMode=N'Exclusive',@LockOwner=N'Session',@LockTimeout=0;
    IF @Lock<0 THROW 51000,N'A US writer/QA owns the operational lock.',1;
    SET @USLock=1;
    EXEC @Lock=HP_MAP_Canada.sys.sp_getapplock @Resource=N'MAP_Operational_Writer',@LockMode=N'Exclusive',@LockOwner=N'Session',@LockTimeout=0;
    IF @Lock<0 THROW 51000,N'A Canada writer/QA owns the operational lock.',1;
    SET @CALock=1;
    EXEC @Lock=HP_MAP_Prod.sys.sp_getapplock @Resource=N'MAP_L12M_Publication',@LockMode=N'Exclusive',@LockOwner=N'Session',@LockTimeout=0;
    IF @Lock<0 THROW 51000,N'An L12M publisher owns the publication lock.',1;
    SET @ReportLock=1;
        SELECT TOP(0) * INTO #MAP_L12M_Ready FROM [_report].[Violations_L12M];
        EXEC [_report].[SP_BUILD_VIOLATIONS_L12M_STAGE]
            @P_Export_Start_Date=@ExportStartDate,@P_Generate_Full_Refresh=@P_Generate_Full_Refresh,@P_Through_Date=@ThroughDate;
        DECLARE @ExpectedRows BIGINT=(SELECT COUNT_BIG(*) FROM #MAP_L12M_Ready);
        BEGIN TRANSACTION;
        DELETE target
        FROM [_report].[Violations_L12M] AS target
        WHERE target.[Date] BETWEEN @ExportStartDate AND @ThroughDate
          AND target.Country IN(N'US',N'Canada');

        INSERT [_report].[Violations_L12M] ([Country],[Validation Status],[Validation Status Weekly],[Fiscal Week Number],[Week],[Date],[Product Category],[Product Subcategory],[Product Subcategory Enforcement],[Fact ID],[SKU],[ASIN],[Product_Description],[Account],[Seller Name],[Storefront Name(s)],[Location ID],[Seller Status],[Termination Date],[Seller Penalty Type],[Reporting Group],[Market],[Log-in Type],[Price Location],[MAP],[Advertised Price],[Eligible IR],[Lowest Possible Price],[Validation_Status_Raw],[Letter Type],[MAP Status Detail],[Letter Delivered Date],[Notification Generation Date],[Buffer Period Start],[Buffer Period End],[Product Suspension Start],[Product Suspension End],[Penalty Period Start],[Penalty Period End],[Screenshot_URL],[DGF Penalty],[Notes],[Tier_Pass_Through],[Category Threshold Highlight],[Category Threshold Alert],[HP_Rep1],[HP_Rep2],[HP_Rep3],[HP_Rep4],[HP_Rep5],[HP_Rep6],[Email1],[Email2],[Email3],[Email4],[Email5],[Email6],[Selector],[Matching Flag],[G20 Sel/Cat Flag],[Weekly Violation Selector],[Validation Status New],[RunningTotal],[UPC],[Bundle Flag],[Minimum Violation Date],[Minimum Listing Date],[Most Recent Listing Date],[Date Requested],[New SKU Flag],[Top_Reseller_Flag],[Latest Courtesy Alert Date],[Login_Type_ID],[Strikethrough Flag],[Weekly_Seller_Rank],[Price Increase SKU Flag],[Content Provider],[Partner_ID],[Violation Type],[Product Category Latest Courtesy Alert Date],[Monitoring Period Start],[Monitoring Period End],[Consequence Penalty Period Start],[Consequence Penalty Period End],[Flag_Warning_Or_Monitoring])
        SELECT [Country],[Validation Status],[Validation Status Weekly],[Fiscal Week Number],[Week],[Date],[Product Category],[Product Subcategory],[Product Subcategory Enforcement],[Fact ID],[SKU],[ASIN],[Product_Description],[Account],[Seller Name],[Storefront Name(s)],[Location ID],[Seller Status],[Termination Date],[Seller Penalty Type],[Reporting Group],[Market],[Log-in Type],[Price Location],[MAP],[Advertised Price],[Eligible IR],[Lowest Possible Price],[Validation_Status_Raw],[Letter Type],[MAP Status Detail],[Letter Delivered Date],[Notification Generation Date],[Buffer Period Start],[Buffer Period End],[Product Suspension Start],[Product Suspension End],[Penalty Period Start],[Penalty Period End],[Screenshot_URL],[DGF Penalty],[Notes],[Tier_Pass_Through],[Category Threshold Highlight],[Category Threshold Alert],[HP_Rep1],[HP_Rep2],[HP_Rep3],[HP_Rep4],[HP_Rep5],[HP_Rep6],[Email1],[Email2],[Email3],[Email4],[Email5],[Email6],[Selector],[Matching Flag],[G20 Sel/Cat Flag],[Weekly Violation Selector],[Validation Status New],[RunningTotal],[UPC],[Bundle Flag],[Minimum Violation Date],[Minimum Listing Date],[Most Recent Listing Date],[Date Requested],[New SKU Flag],[Top_Reseller_Flag],[Latest Courtesy Alert Date],[Login_Type_ID],[Strikethrough Flag],[Weekly_Seller_Rank],[Price Increase SKU Flag],[Content Provider],[Partner_ID],[Violation Type],[Product Category Latest Courtesy Alert Date],[Monitoring Period Start],[Monitoring Period End],[Consequence Penalty Period Start],[Consequence Penalty Period End],[Flag_Warning_Or_Monitoring] FROM #MAP_L12M_Ready;


        IF (SELECT COUNT_BIG(*) FROM [_report].[Violations_L12M] WHERE [Date] BETWEEN @ExportStartDate AND @ThroughDate AND Country IN(N'US',N'Canada'))<>@ExpectedRows
            THROW 51000,N'L12M published row count does not match staging.',1;
        COMMIT TRANSACTION;
    IF @ReportLock=1 EXEC HP_MAP_Prod.sys.sp_releaseapplock @Resource=N'MAP_L12M_Publication',@LockOwner=N'Session';
    IF @CALock=1 EXEC HP_MAP_Canada.sys.sp_releaseapplock @Resource=N'MAP_Operational_Writer',@LockOwner=N'Session';
    IF @USLock=1 EXEC HP_MAP_Prod.sys.sp_releaseapplock @Resource=N'MAP_Operational_Writer',@LockOwner=N'Session';
    END TRY
    BEGIN CATCH
        IF XACT_STATE()<>0 ROLLBACK TRANSACTION;
    IF @ReportLock=1 EXEC HP_MAP_Prod.sys.sp_releaseapplock @Resource=N'MAP_L12M_Publication',@LockOwner=N'Session';
    IF @CALock=1 EXEC HP_MAP_Canada.sys.sp_releaseapplock @Resource=N'MAP_Operational_Writer',@LockOwner=N'Session';
    IF @USLock=1 EXEC HP_MAP_Prod.sys.sp_releaseapplock @Resource=N'MAP_Operational_Writer',@LockOwner=N'Session';
        THROW;
    END CATCH;
    /* MAP_L12M_MAIL_AFTER_COMMIT_V1: queue only after successful publication.
       Mail failure cannot roll back an already committed report or invite an
       automatic retry of the expensive refresh. Queued does not mean delivered. */
    BEGIN TRY
        IF HP_MAP_Prod.dbo.CanSendDbMail() = 1
        BEGIN
            DECLARE @L12M_MailReturn INT, @L12M_MailItemID INT;
            EXEC @L12M_MailReturn = msdb.dbo.sp_send_dbmail
                @profile_name = N'QDN Support',
                @recipients = N'pbelezhkov@dsmengines.com;edeneva@dsmengines.com;zfraile@dsmengines.com',
                @body = N'The _report.Violations_L12M table has been refreshed.',
                @subject = N'The Violations L12M export has been generated',
                @body_format = N'HTML',
                @mailitem_id = @L12M_MailItemID OUTPUT;
            IF ISNULL(@L12M_MailReturn, -1) <> 0
                THROW 51000, N'Database Mail did not return success.', 1;
            RAISERROR(N'L12M publication COMMITTED; success email queued (mailitem_id %d).', 10, 1, @L12M_MailItemID) WITH NOWAIT;
        END;
        ELSE
            RAISERROR(N'L12M publication COMMITTED; email disabled by CanSendDbMail().', 10, 1) WITH NOWAIT;
    END TRY
    BEGIN CATCH
        DECLARE @L12M_MailError NVARCHAR(2048) = ERROR_MESSAGE();
        RAISERROR(N'L12M publication COMMITTED, but its success email was not queued: %s. Investigate mail; do not rerun the refresh only to resend email.', 10, 1, @L12M_MailError) WITH NOWAIT;
    END CATCH;
END;

-- 09/17/26 PB Restored top-seller membership and weekly seller ranking without restoring projected FIN or RunningTotal.
CREATE   PROCEDURE [_report].[SP_BUILD_VIOLATIONS_L12M_STAGE]
    @P_Export_Start_Date DATE = NULL,
    @P_Generate_Full_Refresh BIT = 0,
    @P_Through_Date DATE=NULL
AS
BEGIN
    SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
    DECLARE @BusinessToday DATE=CONVERT(DATE,SYSUTCDATETIME() AT TIME ZONE N'UTC' AT TIME ZONE N'Eastern Standard Time');
    DECLARE @ThroughDate DATE=COALESCE(@P_Through_Date,DATEADD(DAY,-1,@BusinessToday));
    IF @ThroughDate>=@BusinessToday THROW 51000,N'L12M cannot publish the current/future Eastern collection day.',1;
    SET NOCOUNT ON;
    /* MAP_L12M_CALLER_STAGE_V1: no permanent target write in this builder. */
    IF @@TRANCOUNT<>0 THROW 51000,N'L12M build requires no ambient transaction; do not use the old initial publisher.',1;
    IF OBJECT_ID(N'tempdb..#MAP_L12M_Ready') IS NULL
        THROW 51000,N'L12M builder requires its caller-created staging table.',1;
    IF EXISTS(SELECT 1 FROM #MAP_L12M_Ready)
        THROW 51000,N'L12M staging table must be empty.',1;
    DECLARE @CAWriterMode NVARCHAR(32);
    EXEC HP_MAP_Canada.sys.sp_executesql
        N'SELECT @Mode=APPLOCK_MODE(N''public'',N''MAP_Operational_Writer'',N''Session'');',
        N'@Mode NVARCHAR(32) OUTPUT',@Mode=@CAWriterMode OUTPUT;
    IF ISNULL(APPLOCK_MODE(N'public',N'MAP_Operational_Writer',N'Session'),N'NoLock')<>N'Exclusive'
     OR ISNULL(@CAWriterMode,N'NoLock')<>N'Exclusive'
     OR ISNULL(APPLOCK_MODE(N'public',N'MAP_L12M_Publication',N'Session'),N'NoLock')<>N'Exclusive'
        THROW 51000,N'Caller must hold both country writer locks and the L12M publication session lock.',1;
    SET XACT_ABORT ON;

    --Start date
    DECLARE @StartDate DATE =
    (
        SELECT d.Week_Begin_Date_Alt
        FROM HP_MAP_Prod._final.Dim_Date AS d
        WHERE d.Full_Date = DATETRUNC(MONTH, DATEADD(MONTH, -12, @BusinessToday))
    );

    IF @StartDate IS NULL
        THROW 50000, 'The L12M start date is not covered by HP_MAP_Prod._final.Dim_Date.', 1;

    DROP TABLE IF EXISTS #Product_Category;
    CREATE TABLE #Product_Category
    (
        [Product Category] NVARCHAR(255) NOT NULL,
        [Product Subcategory] NVARCHAR(50) NOT NULL,
        [Product Subcategory Enforcement] NVARCHAR(255) NOT NULL,
        [Rule Product Type] NVARCHAR(50) NOT NULL
    );

    INSERT #Product_Category
    (
        [Product Category],
        [Product Subcategory],
        [Product Subcategory Enforcement],
        [Rule Product Type]
    )
    VALUES
        ('Supplies', 'Ink', 'Ink & Toner', 'Ink'),
        ('Supplies', 'Toner', 'Ink & Toner', 'Toner'),
        ('Supplies', 'C-SKU', 'Ink & Toner', 'Toner'),
        ('Supplies', 'G-SKU', 'Ink & Toner', 'Toner'),
        ('Print', 'Inkjet', 'Inkjet', 'Inkjet'),
        ('Print', 'LaserJet', 'LaserJet', 'LaserJet'),
        ('Large Format', 'Large Format', 'Large Format', 'Large Format'),
        ('Large Format', 'Large Format Supplies', 'Large Format', 'Large Format Supplies'),
        ('PS', 'BPS Display', 'BPS Display', 'BPS Display'),
        ('PS', 'CPS Desktop', 'CPS Desktop', 'CPS Desktop'),
        ('PS', 'CPS Display', 'CPS Display', 'CPS Display'),
        ('PS', 'CPS Notebook', 'CPS Notebook', 'CPS Notebook');

    DROP TABLE IF EXISTS #Weeks;
    RAISERROR(N'L12M stage starting #Weeks',10,1) WITH NOWAIT;
    SELECT DISTINCT
        dt.Week_Begin_Date_Alt AS [Week],
        dt.Full_Date AS [Date]
    INTO #Weeks
    FROM HP_MAP_Prod._final.Dim_Date AS dt  -- 9/10/20 LU adjusted to Dim_Date to include days even if no data loaded.
    WHERE dt.Week_Begin_Date_Alt >= @StartDate
      AND dt.Full_Date < DATEADD(DAY,1,@ThroughDate);

    CREATE UNIQUE CLUSTERED INDEX IX_Weeks_Date ON #Weeks ([Date]);

    DROP TABLE IF EXISTS #UPC_US;
    DECLARE @MaxFileDateUPC_US DATE =
    (
        SELECT MAX([File Date])
        FROM HP_MAP_Prod._analysis.MAP_Price_List_Tool_Export_NEW
    );

    RAISERROR(N'L12M stage starting #UPC_US',10,1) WITH NOWAIT;
    SELECT
        SKU,
        MAX(ISNULL(UPC, '')) AS UPC
    INTO #UPC_US
    FROM HP_MAP_Prod._analysis.MAP_Price_List_Tool_Export_NEW
    WHERE [File Date] = @MaxFileDateUPC_US
    GROUP BY SKU;

    CREATE UNIQUE CLUSTERED INDEX IX_UPC_US_SKU ON #UPC_US (SKU);

    DROP TABLE IF EXISTS #MinViolationDate;
    RAISERROR(N'L12M stage starting #MinViolationDate',10,1) WITH NOWAIT;
    SELECT
        New_Location_ID,
        MIN(Collected_Date) AS [Minimum Violation Date]
    INTO #MinViolationDate
    FROM [_final].[ViolationsMAP]
    WHERE Collected_Date >= '2023-11-01'
    GROUP BY New_Location_ID;

    CREATE UNIQUE CLUSTERED INDEX IX_MinViolationDate_Location
        ON #MinViolationDate (New_Location_ID);

    /* Fact_Listings is retained only for the two explicitly preserved Tableau
       fields. It is no longer used to recreate or supplement violations. */
    DROP TABLE IF EXISTS #Min_Max_Listing_Date;
    RAISERROR(N'L12M stage starting #Min_Max_Listing_Date',10,1) WITH NOWAIT;
    SELECT
        s.Master_New_Location_Id,
        MIN(CAST(f.Collected_Date AS DATE)) AS [Minimum Listing Date],
        MAX(CAST(f.Collected_Date AS DATE)) AS [Most Recent Listing Date]
    INTO #Min_Max_Listing_Date
    FROM HP_MAP_Prod._final.Fact_Listings AS f
    INNER JOIN HP_MAP_Prod._final.Dim_Seller AS s
        ON s.Seller_ID = f.Seller_ID
    WHERE CAST(f.Collected_Date AS DATE) >= '2023-11-01'
      AND s.Master_New_Location_Id <> 'N/A'
    GROUP BY s.Master_New_Location_Id;

    CREATE UNIQUE CLUSTERED INDEX IX_MinMaxListingDate_Location
        ON #Min_Max_Listing_Date (Master_New_Location_Id);

    DROP TABLE IF EXISTS #DateRequested;
    RAISERROR(N'L12M stage starting #DateRequested',10,1) WITH NOWAIT;
    SELECT
        Country AS Ctry,
        Master_New_Location_Id,
        MIN(Date_Requested) AS [Date Requested]
    INTO #DateRequested
    FROM HP_MAP_Prod._analysis.PM_Monitoring_List
    GROUP BY Country, Master_New_Location_Id;

    CREATE NONCLUSTERED INDEX IX_DateRequested_Country_Location

                +CONVERT(NVARCHAR(10),@MAP_L12M_US_BatchStart,120)+N'; accumulated rows '

            ORDER BY base.Flag_Triggered_Warning DESC,
                REPLACE(ISNULL(base.Validation_Status_Raw,N'Z'),N'Not Inspected',N'Z'),
                base.Market_ID,base.Violation_ID) AS Violation_ID_Selector

        CONVERT
        (
            VARCHAR(25),
            CASE WHEN ISNULL(violation.Flag_StrikeThrough, 0) = 1
                 THEN 'Strikethrough Pricing' ELSE 'Non-Strikethrough Pricing' END
        ) AS [Strikethrough Flag],
        CONVERT
        (
            INT,
            CASE
                WHEN ISNULL(violation.Flag_Bundle, 0) = 0
                 AND EXISTS
                 (
                     SELECT 1
                     FROM #Grace_Period AS gps
                     WHERE gps.Country = 'US'
                       AND gps.SKU = violation.SKU
                       AND scaffold.[Date] BETWEEN gps.Valid_From AND gps.Valid_To
                 ) THEN 1
                WHEN ISNULL(violation.Flag_Bundle, 0) = 1
                 AND EXISTS
                 (
                     SELECT 1
                     FROM #Grace_Period AS gps_bundle
                     WHERE gps_bundle.Country = 'US'
                       AND violation.SKU LIKE '%' + gps_bundle.SKU + '%'
                       AND scaffold.[Date] BETWEEN gps_bundle.Valid_From AND gps_bundle.Valid_To
                 ) THEN 1
                ELSE 0
            END
        ) AS [Price Increase SKU Flag],
        CONVERT(NVARCHAR(255), LEFT(content.Content_Source, 255)) AS [Content Provider],
        partner.Partner_ID,
        CONVERT(NVARCHAR(20), LEFT(ISNULL(violation.Violation_Type, 'N/A'), 20)) AS [Violation Type],
        CAST(NULL AS DATE) AS [Product Category Latest Courtesy Alert Date],
        warning_period.Penalty_Period_Start_Date AS [Monitoring Period Start],
        warning_period.Penalty_Period_End_Date_Adjusted AS [Monitoring Period End],
        consequence_period.Penalty_Period_Start_Date AS [Consequence Penalty Period Start],
        consequence_period.Penalty_Period_End_Date_Adjusted AS [Consequence Penalty Period End],
        CONVERT
        (
            INT,
            CASE
                WHEN violation.Flag_Triggered_Warning = 1 THEN 2
                WHEN warning_period.ID IS NOT NULL THEN 1
                ELSE 0
            END
        ) AS Flag_Warning_Or_Monitoring
    INTO #OutputBase
    FROM #CTE_cj AS scaffold
    LEFT JOIN #ViolationRows AS violation
        ON violation.[Location ID] = scaffold.[Location ID]
       AND violation.Master_SID = scaffold.Master_SID --rv 4/12/23
       AND violation.[Product Subcategory] = scaffold.[Product Subcategory]
       AND violation.[Date] = scaffold.[Date]
    LEFT JOIN #Fiscal_Week AS fiscal
        ON fiscal.Week_Begin_Date_Alt = scaffold.[Week]
    LEFT JOIN #TopResellers AS top_seller
        ON top_seller.New_Location_ID = scaffold.[Location ID]
    OUTER APPLY
    (
        SELECT TOP (1)
            ds.Supplies_Qualification,
            ds.Print_Qualification,
            ds.Qualified_Business_Personal_Systems,
            ds.Qualified_Consumer_Personal_Systems
        FROM HP_MAP_Prod._final.Dim_Seller AS ds
        WHERE ds.Master_SID = scaffold.Master_SID
          AND scaffold.[Date] BETWEEN ds.Valid_From AND ds.Valid_To
        ORDER BY ds.Valid_From DESC, ds.Seller_ID DESC
    ) AS seller_at_date
    LEFT JOIN #ActiveQPP AS qpp
        ON qpp.HPI_Location_ID = scaffold.[Location ID]
    LEFT JOIN #LatestNotifications AS latest_notification
        ON latest_notification.New_Location_ID = scaffold.[Location ID]
       AND latest_notification.Product_Type = scaffold.[Product Subcategory Enforcement]
    LEFT JOIN #LatestStrikes AS latest_strike
        ON latest_strike.New_Location_ID = scaffold.[Location ID]
       AND latest_strike.Product_Type = scaffold.[Product Subcategory Enforcement]
    /* Historical-period approach: inactive/rehabilitated strikes remain valid
       evidence that an older report date was inside a Warning monitoring period. */
    OUTER APPLY
    (
        SELECT TOP (1)
            s.ID,
            s.Violation_Status,
            s.Penalty_Period_Start_Date,
            s.Penalty_Period_End_Date_Adjusted
        FROM #ValidStrikes AS s
        WHERE s.New_Location_ID = scaffold.[Location ID]
          AND s.Product_Type = scaffold.[Product Subcategory Enforcement]
          AND LEFT(s.Violation_Status, 1) = 'W'
          AND scaffold.[Date] BETWEEN s.Penalty_Period_Start_Date
                                  AND s.Penalty_Period_End_Date_Adjusted
        ORDER BY s.Penalty_Period_Start_Date DESC, s.ID DESC
    ) AS warning_period
    OUTER APPLY
    (
        SELECT TOP (1)
            s.ID,
            s.Violation_Status,
            s.Penalty_Period_Start_Date,
            s.Penalty_Period_End_Date_Adjusted
        FROM #ValidStrikes AS s
        WHERE s.New_Location_ID = scaffold.[Location ID]
          AND s.Product_Type = scaffold.[Product Subcategory Enforcement]
          AND LEFT(s.Violation_Status, 1) = 'C'
          AND scaffold.[Date] BETWEEN s.Penalty_Period_Start_Date
                                  AND s.Penalty_Period_End_Date_Adjusted
        ORDER BY s.Penalty_Period_Start_Date DESC, s.ID DESC
    ) AS consequence_period
    OUTER APPLY
    (
        SELECT
            CASE
                WHEN warning_period.ID IS NOT NULL THEN
                    CASE warning_period.Violation_Status
                        WHEN 'WF' THEN 'CF'
                        WHEN 'W30' THEN 'C30'
                        WHEN 'W60' THEN 'C60'
                        WHEN 'W90' THEN 'C90'
                        ELSE 'C30'
                    END
                WHEN COALESCE(violation.[Seller Penalty Type], scaffold.[Seller Penalty Type]) LIKE '%Financial'
                    THEN 'WF'
                WHEN consequence_period.Violation_Status = 'C30' THEN 'W60'
                WHEN consequence_period.Violation_Status IN ('C60', 'C90') THEN 'W90'
                ELSE 'W30'
            END AS Rule_Action
    ) AS threshold_action
    OUTER APPLY
    (
        SELECT TOP (1)
            r.Threshold_Action
        FROM [_ref].[Violation_Rules] AS r
        WHERE r.Program = 'MAP'
          AND r.Seller_Group IN
              (COALESCE(violation.[Reporting Group], scaffold.[Reporting Group], 'All'), 'All')
          AND r.Penalty_Description = COALESCE(violation.[Seller Penalty Type], scaffold.[Seller Penalty Type])
          AND r.Product_Type_Hybrid = scaffold.[Rule Product Type]
          AND r.[Action] = threshold_action.Rule_Action
          AND scaffold.[Date] BETWEEN r.Valid_From AND r.Valid_To
        ORDER BY
            CASE WHEN r.Seller_Group = COALESCE(violation.[Reporting Group], scaffold.[Reporting Group], 'All') THEN 0 ELSE 1 END,
            r.Valid_From DESC,
            r.ID DESC
    ) AS threshold_rule
    OUTER APPLY
    (
        SELECT
            CASE
                WHEN warning_period.ID IS NULL THEN 1
                WHEN scaffold.[Product Subcategory] IN ('Ink', 'Toner', 'C-SKU', 'G-SKU', 'Large Format Supplies') THEN 5
                ELSE 2
            END AS Threshold_Action
    ) AS threshold_fallback
    LEFT JOIN #CTE_Contacts AS contacts
        ON contacts.[New Location ID] = scaffold.[Location ID]  -- 9/10/20 LU added for G20 Contacts
       AND contacts.[Product Category] =
           CASE
               WHEN scaffold.[Product Subcategory] LIKE 'Large Format%' THEN 'Large Format'
               WHEN scaffold.[Product Subcategory] IN ('C-SKU', 'G-SKU') THEN 'Supplies' --DP 1/12/23 adjusting join so Supplies contacts apply to C-SKU in tooltip
               ELSE scaffold.[Product Category]
           END
    LEFT JOIN #UPC_US AS upc
        ON upc.SKU = violation.SKU
    LEFT JOIN #MinViolationDate AS min_violation
        ON min_violation.New_Location_ID = scaffold.[Location ID]
    LEFT JOIN #Min_Max_Listing_Date AS listing_dates
        ON listing_dates.Master_New_Location_Id = scaffold.[Location ID]
    LEFT JOIN #DateRequested AS date_requested
        ON date_requested.Master_New_Location_Id = scaffold.[Location ID]
       AND date_requested.Ctry = 'US'
    LEFT JOIN #NewSKU AS new_sku
        ON new_sku.SKU = violation.SKU
    LEFT JOIN #Content AS content
        ON content.HPI_Location_ID = scaffold.[Location ID]
       AND content.Country IN ('US', 'USA')

        ON partner.LocationId = scaffold.[Location ID]
    WHERE scaffold.[Location ID] <> 'N/A'; -- 10/14/24 ED

    LEFT JOIN #CA_ViolationRows AS violation
        ON violation.[Location ID] = scaffold.[Location ID]

       AND violation.[Product Subcategory] = scaffold.[Product Subcategory]

          AND scaffold.[Date] BETWEEN ds.Valid_From AND ds.Valid_To
        ORDER BY ds.Valid_From DESC, ds.Seller_ID DESC
    ) AS seller_at_date
    LEFT JOIN #CA_ActiveQPP AS qpp
        ON qpp.HPI_Location_ID = scaffold.[Location ID]
    LEFT JOIN #CA_LatestNotifications AS latest_notification
        ON latest_notification.New_Location_ID = scaffold.[Location ID]
       AND latest_notification.Product_Type = scaffold.[Product Subcategory Enforcement]
    LEFT JOIN #CA_LatestStrikes AS latest_strike
        ON latest_strike.New_Location_ID = scaffold.[Location ID]
       AND latest_strike.Product_Type = scaffold.[Product Subcategory Enforcement]
    /* Historical-period approach: inactive/rehabilitated strikes remain valid
       evidence that an older report date was inside a Warning monitoring period. */
    OUTER APPLY
    (
        SELECT TOP (1)
            s.ID,
            s.Violation_Status,
            s.Penalty_Period_Start_Date,
            s.Penalty_Period_End_Date_Adjusted
        FROM #CA_ValidStrikes AS s
        WHERE s.New_Location_ID = scaffold.[Location ID]
          AND s.Product_Type = scaffold.[Product Subcategory Enforcement]
          AND LEFT(s.Violation_Status, 1) = N'W'
          AND scaffold.[Date] BETWEEN s.Penalty_Period_Start_Date
                                  AND s.Penalty_Period_End_Date_Adjusted

    OUTER APPLY
    (
        SELECT TOP (1)
            s.ID,
            s.Violation_Status,
            s.Penalty_Period_Start_Date,
            s.Penalty_Period_End_Date_Adjusted
        FROM #CA_ValidStrikes AS s
        WHERE s.New_Location_ID = scaffold.[Location ID]
          AND s.Product_Type = scaffold.[Product Subcategory Enforcement]
          AND LEFT(s.Violation_Status, 1) = N'C'
          AND scaffold.[Date] BETWEEN s.Penalty_Period_Start_Date
                                  AND s.Penalty_Period_End_Date_Adjusted
        ORDER BY s.Penalty_Period_Start_Date DESC, s.ID DESC

    OUTER APPLY
    (
        SELECT
            CASE
                WHEN warning_period.ID IS NOT NULL THEN
                    CASE warning_period.Violation_Status
                        WHEN N'WF' THEN N'CF'
                        WHEN N'W30' THEN N'C30'
                        WHEN N'W60' THEN N'C60'
                        WHEN N'W90' THEN N'C90'
                        ELSE N'C30'
                    END
                WHEN COALESCE(violation.[Seller Penalty Type], scaffold.[Seller Penalty Type]) LIKE N'%Financial'
                    THEN N'WF'
                WHEN consequence_period.Violation_Status = N'C30' THEN N'W60'
                WHEN consequence_period.Violation_Status IN (N'C60', N'C90') THEN N'W90'
                ELSE N'W30'
            END AS Rule_Action
    ) AS threshold_action
    OUTER APPLY
    (
        SELECT TOP (1)
            r.Threshold_Action
        FROM [HP_MAP_Canada].[_ref].[Violation_Rules] AS r
        WHERE r.Program = N'MAP'
          AND r.Seller_Group IN
              (COALESCE(violation.[Reporting Group], scaffold.[Reporting Group], N'All'), N'All')
          AND r.Penalty_Description = COALESCE(violation.[Seller Penalty Type], scaffold.[Seller Penalty Type])
          AND r.Product_Type_Hybrid = scaffold.[Rule Product Type]
          AND r.[Action] = threshold_action.Rule_Action
          AND scaffold.[Date] BETWEEN r.Valid_From AND r.Valid_To
        ORDER BY
            CASE WHEN r.Seller_Group = COALESCE(violation.[Reporting Group], scaffold.[Reporting Group], N'All') THEN 0 ELSE 1 END,
            r.Valid_From DESC,
            r.ID DESC
    ) AS threshold_rule
    OUTER APPLY
    (
        SELECT
            CASE
                WHEN warning_period.ID IS NULL THEN 1
                WHEN scaffold.[Product Subcategory] IN (N'Ink', N'Toner', N'C-SKU', N'G-SKU')
                    THEN CASE WHEN COALESCE(violation.[Seller Penalty Type],scaffold.[Seller Penalty Type]) LIKE N'%Financial%' THEN 10 ELSE 5 END

    ) AS threshold_fallback
    LEFT JOIN #CA_CTE_Contacts AS contacts
        ON contacts.[New Location ID] = scaffold.[Location ID]  -- 9/10/20 LU added for G20 Contacts
       AND contacts.[Product Category] =
           CASE
               WHEN scaffold.[Product Subcategory] LIKE N'Large Format%' THEN N'Large Format'
               WHEN scaffold.[Product Subcategory] IN (N'C-SKU', N'G-SKU') THEN N'Supplies' --DP 1/12/23 adjusting join so Supplies contacts apply to C-SKU in tooltip
               ELSE scaffold.[Product Category]
           END
    LEFT JOIN #CA_UPC_US AS upc
        ON upc.SKU = violation.SKU
    LEFT JOIN #CA_MinViolationDate AS min_violation
        ON min_violation.New_Location_ID = scaffold.[Location ID]
    LEFT JOIN #CA_Min_Max_Listing_Date AS listing_dates
        ON listing_dates.Master_New_Location_Id = scaffold.[Location ID]
    LEFT JOIN #CA_DateRequested AS date_requested
        ON date_requested.Master_New_Location_Id = scaffold.[Location ID]
       AND date_requested.Ctry = N'Canada'
    LEFT JOIN #CA_NewSKU AS new_sku
        ON new_sku.SKU = violation.SKU
    LEFT JOIN #CA_Content AS content
        ON content.HPI_Location_ID = scaffold.[Location ID]
       AND content.Country IN (N'CA', N'Canada')

                REPLACE(ISNULL(output_base.[Validation Status], N'Z'), N'Not Inspected', N'Z'),

        ) AS [Weekly Violation Selector] -- 9/17/2020 LU: Moved up a line, certain weeks weren't getting a WVS of 1
    INTO #CA_FinalOutput
    FROM #CA_OutputBase AS output_base
    INNER JOIN #CA_WeeklySellerRanks AS seller_rank
        ON seller_rank.Country = output_base.Country
       AND seller_rank.[Week] = output_base.[Week]
       AND seller_rank.[Product Subcategory Enforcement] = output_base.[Product Subcategory Enforcement]
       AND seller_rank.[Location ID] = output_base.[Location ID];

		AND mpa.Country = 'US'

	LEFT JOIN _ref.Seller_Market_Mapping smm WITH(NOLOCK) 
		ON brs.Seller_Name_Unified = smm.Seller_Name_Unified			--CH 7/16/21: adding new marketplace tables to bring in QC

	WHERE

			AND mkt.In_MAP_Scope_Flag = 'In MAP Scope'

									AND v.[Location ID] = sel.Master_New_Location_Id	--CH 11/4/19 added to account for HP; not in ViolationsMAP

DROP TABLE IF EXISTS #CTE_Union --combining each of the three, then applying derived VI results
SELECT CASE WHEN [Validation_Status_Raw] = 'Not Inspected'
			THEN CASE WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Accepted' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU,Login_Type_ID) = 1 THEN 'Accepted - Derived'
					WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Rejected' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU,Login_Type_ID) = 1 THEN 'Rejected - Derived' 
					ELSE Validation_Status_Raw END
		ELSE [Validation_Status_Raw]
		END [Validation Status]
	,CASE WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Accepted' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU,Login_Type_ID) = 1 THEN 'Accepted'
					WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Rejected' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU,Login_Type_ID) = 1 THEN 'Rejected' 
					ELSE Validation_Status_Raw END [Validation Status Weekly]
	,a.*
INTO #CTE_Union
FROM (
	SELECT *
		,'N/A' AS Violation_Type
		,DENSE_RANK() OVER(ORDER BY Week DESC) [Adhoc Week Selector]
	FROM #CTE_Adhoc

	UNION

	SELECT * 
		,0 
	FROM #CTE_Vtable

	UNION 

	SELECT *
		,'N/A' AS Violation_Type
		,-1 
	FROM #CTE_Fact
) a

DROP TABLE IF EXISTS #ca_CTE_Union  --combining each of the three, then applying derived VI results
SELECT CASE WHEN [Validation_Status_Raw] = 'Not Inspected'
			THEN CASE WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Accepted' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU,ISNULL(Login_Type_ID, 1)) = 1 THEN 'Accepted - Derived'
					WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Rejected' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU,ISNULL(Login_Type_ID, 1)) = 1 THEN 'Rejected - Derived' 
					ELSE Validation_Status_Raw END
		ELSE [Validation_Status_Raw]
		END [Validation Status]
	,CASE WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Accepted' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU) = 1 THEN 'Accepted'
					WHEN MAX(CASE WHEN [Validation_Status_Raw] = 'Rejected' THEN 1 ELSE 0 END) OVER(PARTITION BY [Week],[Location ID],SKU) = 1 THEN 'Rejected' 
					ELSE Validation_Status_Raw END [Validation Status Weekly]
	,a.*
INTO #ca_CTE_Union
FROM (
	SELECT *
		,'N/A' AS Violation_Type
		,DENSE_RANK() OVER(ORDER BY Week DESC) [Adhoc Week Selector]
	FROM #ca_CTE_Adhoc WITH(NOLOCK) 

	UNION 
		
	SELECT * 
		,0 [Adhoc Week Selector]
	FROM #ca_CTE_Vtable WITH(NOLOCK) 

	UNION 

	SELECT *
		,'N/A' AS Violation_Type
		,-1 
	FROM #ca_CTE_Fact WITH(NOLOCK)
) a

	,[Location ID]
	,Master_SID
